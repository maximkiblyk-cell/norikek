cmake_minimum_required(VERSION 3.20)
project(nori_game_core VERSION 0.8.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NORI_BUILD_HOST "Build the development-only host" ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
if(WIN32)
 find_package(CURL CONFIG REQUIRED)
 set(NORI_CURL_TARGET CURL::libcurl)
else()
 set(NORI_CURL_STATIC_ARCHIVE "" CACHE FILEPATH "Explicit static libcurl archive; requires libcurl.pc")
 if(DEFINED CURL_LIBRARY AND CURL_LIBRARY MATCHES "\\.a$")
  message(FATAL_ERROR "Do not pass a static CURL_LIBRARY directly; use NORI_CURL_STATIC_ARCHIVE with matching libcurl.pc")
 endif()
 if(NORI_CURL_STATIC_ARCHIVE)
  if(NOT EXISTS "${NORI_CURL_STATIC_ARCHIVE}" OR NOT NORI_CURL_STATIC_ARCHIVE MATCHES "\\.a$")
   message(FATAL_ERROR "NORI_CURL_STATIC_ARCHIVE must name an existing .a archive")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(NORI_CURL_PC REQUIRED libcurl)
  execute_process(COMMAND "${PKG_CONFIG_EXECUTABLE}" --static --libs libcurl RESULT_VARIABLE pkg_status OUTPUT_VARIABLE curl_static_flags OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT pkg_status EQUAL 0)
   message(FATAL_ERROR "libcurl.pc static dependency metadata is required")
  endif()
  separate_arguments(curl_static_flags NATIVE_COMMAND "${curl_static_flags}")
  list(FILTER curl_static_flags EXCLUDE REGEX "^-l(curl|:libcurl\\.a)$")
  add_library(nori_curl_static INTERFACE)
  target_link_libraries(nori_curl_static INTERFACE "${NORI_CURL_STATIC_ARCHIVE}" ${curl_static_flags})
  set(NORI_CURL_TARGET nori_curl_static)
 else()
  # Never silently select a static archive without transitive dependencies.
  set(saved_suffixes "${CMAKE_FIND_LIBRARY_SUFFIXES}")
  if(APPLE)
   set(CMAKE_FIND_LIBRARY_SUFFIXES .dylib)
  else()
   set(CMAKE_FIND_LIBRARY_SUFFIXES .so .so.4)
  endif()
  find_library(NORI_CURL_SHARED_LIBRARY NAMES curl libcurl.so.4 REQUIRED)
  set(CMAKE_FIND_LIBRARY_SUFFIXES "${saved_suffixes}")
  if(NORI_CURL_SHARED_LIBRARY MATCHES "\\.a$")
   message(FATAL_ERROR "NORI_CURL_SHARED_LIBRARY cannot be static; use NORI_CURL_STATIC_ARCHIVE")
  endif()
  set(NORI_CURL_TARGET "${NORI_CURL_SHARED_LIBRARY}")
 endif()
endif()
add_library(nori_core STATIC
 src/core/core.cpp src/navigation/navigation.cpp src/input/input.cpp src/settings/settings.cpp
 src/ui/model.cpp src/network/http.cpp src/authentication/auth.cpp src/authentication/session.cpp src/authentication/controller.cpp src/licensing/license.cpp
 src/logging/logger.cpp src/notifications/notifications.cpp src/adapters/adapters.cpp src/platform/vault.cpp)
target_include_directories(nori_core PUBLIC include third_party/json PRIVATE third_party/curl)
target_link_libraries(nori_core PUBLIC ${NORI_CURL_TARGET} OpenSSL::Crypto Threads::Threads)
if(WIN32)
 target_link_libraries(nori_core PRIVATE Crypt32)
endif()
if(MSVC)
 target_compile_options(nori_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(nori_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

target_link_libraries(nori_http_tests PRIVATE nori_core)
